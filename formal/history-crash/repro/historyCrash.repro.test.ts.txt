/**
 * Deterministic repros for the TLC findings in formal/history-crash/ (cases in check.sh).
 *
 * NOT part of CI. Each test asserts the CORRECT (crash-consistent) behavior, so on the
 * current code every test FAILS; a FAIL means "bug reproduced". Run with:
 *
 *   cp formal/history-crash/repro/historyCrash.repro.test.ts.txt \
 *      src/node/services/historyCrash.repro.test.ts
 *   bun test src/node/services/historyCrash.repro.test.ts
 *   rm src/node/services/historyCrash.repro.test.ts
 *
 * A crash is injected by making a path-specific filesystem call fail (the in-code error
 * handling then leaves exactly the on-disk state a process crash at that point would
 * leave), by stopping the call sequence, or (F5) by SIGKILLing a real second backend
 * process. Reload = a fresh HistoryService on the same root (empty in-memory state).
 */
import { afterEach, describe, expect, spyOn, test } from "bun:test";
import assert from "node:assert";
import { spawnSync } from "node:child_process";
import * as fs from "fs/promises";
import nodeFs from "fs";
import * as path from "path";
import { createMuxMessage, type MuxMessage } from "@/common/types/message";
import type { Result } from "@/common/types/result";
import { HistoryService } from "./historyService";
import { createTestHistoryService } from "./testHistoryService";

const cleanups: Array<() => Promise<void>> = [];
afterEach(async () => {
  for (const cleanup of cleanups.splice(0)) await cleanup();
});

async function setup() {
  const created = await createTestHistoryService();
  cleanups.push(created.cleanup);
  return created;
}

function ok<T>(result: Result<T>): T {
  assert(result.success, result.success ? "" : result.error);
  return result.data;
}

async function activeIds(service: HistoryService, ws: string): Promise<string[]> {
  return ok(await service.getHistoryFromLatestBoundary(ws)).map((m) => m.id);
}

async function fullIds(service: HistoryService, ws: string): Promise<string[]> {
  const ids: string[] = [];
  ok(
    await service.iterateFullHistory(ws, "forward", (chunk) => {
      ids.push(...chunk.map((m) => m.id));
    })
  );
  return ids;
}

async function seqOf(service: HistoryService, ws: string, id: string): Promise<number> {
  const row = ok(await service.getHistoryFromLatestBoundary(ws)).find((m) => m.id === id);
  const seq = row?.metadata?.historySequence;
  assert(typeof seq === "number", `no historySequence for ${id}`);
  return seq;
}

function assistant(id: string, text: string, historySequence: number): MuxMessage {
  return createMuxMessage(id, "assistant", text, { historySequence });
}

/** A turn up to its first partial flush: user row, empty placeholder, partial.json. */
async function startTurn(service: HistoryService, ws: string, userId: string, asstId: string) {
  ok(await service.appendToHistory(ws, createMuxMessage(userId, "user", `prompt ${userId}`)));
  ok(await service.appendToHistory(ws, createMuxMessage(asstId, "assistant", ""))); // turnRequestBuilder.ts:3160
  const seq = await seqOf(service, ws, asstId);
  ok(await service.writePartial(ws, assistant(asstId, `streamed ${asstId}`, seq)));
  return seq;
}

describe("F3: stream completion deletes partial.json before updateHistory", () => {
  // streamManager.ts:4935 deletePartial, then 4943 updateHistory(final).
  test("a crash between the two calls loses the completed response", async () => {
    const { historyService: s1, config } = await setup();
    const ws = "ws-f3-crash";
    await startTurn(s1, ws, "u0", "a0");
    ok(await s1.deletePartial(ws)); // streamManager.ts:4935
    // Process dies before streamManager.ts:4943.
    const s2 = new HistoryService(config);
    const partial = await s2.readPartial(ws);
    const row = ok(await s2.getHistoryFromLatestBoundary(ws)).find((m) => m.id === "a0");
    const recoverable = partial?.id === "a0" || (row?.parts.length ?? 0) > 0;
    expect(recoverable).toBe(true);
  });

  test("an updateHistory I/O failure (no crash) also loses it", async () => {
    const { historyService: s1, config } = await setup();
    const ws = "ws-f3-eio";
    const seq = await startTurn(s1, ws, "u0", "a0");
    ok(await s1.deletePartial(ws));
    const original = nodeFs.rename;
    const renameSpy = spyOn(nodeFs, "rename").mockImplementation(((
      from: nodeFs.PathLike,
      to: nodeFs.PathLike,
      callback: nodeFs.NoParamCallback
    ) => {
      if (String(to).endsWith("chat.jsonl")) {
        callback(Object.assign(new Error("EIO: injected"), { code: "EIO" }));
        return;
      }
      original(from, to, callback);
    }) as typeof nodeFs.rename);
    const update = await s1.updateHistory(ws, assistant("a0", "final answer", seq));
    renameSpy.mockRestore();
    expect(update.success).toBe(false); // streamManager.ts:4947 only logs this
    const s2 = new HistoryService(config);
    const partial = await s2.readPartial(ws);
    const row = ok(await s2.getHistoryFromLatestBoundary(ws)).find((m) => m.id === "a0");
    expect(partial?.id === "a0" || (row?.parts.length ?? 0) > 0).toBe(true);
  });
});

describe("F1: edit truncation publishes chat.jsonl before retiring partial.json", () => {
  test("a crash between publish (5443) and retirement (5478) resurrects the discarded turn", async () => {
    const { historyService: s1, config } = await setup();
    const ws = "ws-f1";
    await startTurn(s1, ws, "u0", "a0");
    // Crash #1: mid-stream; partial.json stays behind (normal crash recovery data).
    const s2 = new HistoryService(config);
    const partialPath = path.join(path.join(config.sessionsDir, ws), "partial.json");
    const original = fs.unlink;
    const unlinkSpy = spyOn(fs, "unlink").mockImplementation(async (target) => {
      // Crash #2: the process dies right after the truncated chat.jsonl was published.
      if (String(target) === partialPath) throw Object.assign(new Error("crash"), { code: "EIO" });
      return original(target);
    });
    const cut = await s2.truncateAfterMessage(ws, "u0"); // user edits u0
    unlinkSpy.mockRestore();
    ok(cut);
    expect(ok(cut).removedMessages.map((m) => m.id)).toContain("a0");
    // Restart; the user re-sends the edited prompt (agentSession.ts:7656 / aiService.ts:970).
    const s3 = new HistoryService(config);
    ok(await s3.appendToHistory(ws, createMuxMessage("u0-edited", "user", "edited prompt")));
    ok(await s3.commitPartial(ws));
    expect(await activeIds(s3, ws)).not.toContain("a0");
  });
});

describe("F2: a second backend edits while the first one streams", () => {
  async function twoBackends() {
    const { historyService: a, config } = await setup();
    const b = new HistoryService(config); // second backend on the same root
    return { a, b, config };
  }

  test("the streaming backend's abort commit fails (stale historySequence) and blocks sends", async () => {
    const { a, b } = await twoBackends();
    const ws = "ws-f2-err";
    const seq = await startTurn(a, ws, "u0", "a0");
    ok(await b.truncateAfterMessage(ws, "u0")); // retires partial.json
    ok(await a.writePartial(ws, assistant("a0", "streamed a0 more", seq))); // next delta recreates it
    const commit = await a.commitPartial(ws, "a0"); // streamManager.ts:2350
    expect(commit.success).toBe(true);
  });

  test("the recreated partial of the discarded turn is committed by the next send", async () => {
    const { a, b } = await twoBackends();
    const ws = "ws-f2-ghost";
    const seq = await startTurn(a, ws, "u0", "a0");
    ok(await b.truncateAfterMessage(ws, "u0"));
    ok(await a.writePartial(ws, assistant("a0", "streamed a0 more", seq)));
    await a.commitPartial(ws, "a0"); // Err on current code; partial.json is retained
    ok(await b.appendToHistory(ws, createMuxMessage("u1", "user", "new prompt")));
    ok(await b.commitPartial(ws)); // stream start in backend B
    expect(await activeIds(b, ws)).not.toContain("a0");
  });

  test("stream completion overwrites a foreign row that reused the removed historySequence", async () => {
    const { a, b } = await twoBackends();
    const ws = "ws-f2-overwrite";
    const seq = await startTurn(a, ws, "u0", "a0");
    ok(await b.truncateAfterMessage(ws, "u0"));
    ok(await b.appendToHistory(ws, createMuxMessage("u1", "user", "new prompt")));
    ok(await b.appendToHistory(ws, createMuxMessage("a1", "assistant", "")));
    expect(await seqOf(b, ws, "a1")).toBe(seq); // sequence reused after the cut
    ok(await a.deletePartial(ws)); // streamManager.ts:4935
    await a.updateHistory(ws, assistant("a0", "final a0", seq)); // streamManager.ts:4943
    const ids = await activeIds(b, ws);
    expect(ids).toContain("a1");
    expect(ids).not.toContain("a0");
  });
});

describe("F5: read-path rotation skips truncate recovery", () => {
  test("rotation over a crashed foreign truncation loses sealed rows", async () => {
    const { historyService: seed, config } = await setup();
    const ws = "ws-f5";
    const dir = path.join(config.sessionsDir, ws);
    const row = (id: string, seq: number, epoch?: number) =>
      JSON.stringify({
        ...createMuxMessage(
          id,
          epoch ? "assistant" : "user",
          epoch ? `summary ${epoch}` : `message ${id}`,
          epoch
            ? {
                historySequence: seq,
                compactionBoundary: true,
                compacted: "user",
                compactionEpoch: epoch,
              }
            : { historySequence: seq }
        ),
        workspaceId: ws,
      }) + "\n";
    await fs.mkdir(dir, { recursive: true });
    await fs.writeFile(
      path.join(dir, "chat.jsonl"),
      row("pre-0", 0) + row("boundary-1", 1, 1) + row("post-0", 2)
    );
    ok(await seed.getHistoryFromLatestBoundary(ws)); // rotates pre-0 into the archive
    // A second boundary whose rotation did not happen yet (crash between boundary write and
    // rotation, a state ensureSealedHistoryRotatedUnlocked documents and heals lazily).
    await fs.appendFile(path.join(dir, "chat.jsonl"), row("boundary-2", 3, 2) + row("post-1", 4));
    const before = await fullIds(new HistoryService(config), ws);
    expect(before).toEqual(["pre-0", "boundary-1", "post-0", "boundary-2", "post-1"]);

    // Backend A (fresh process state). Its read probe (1430) runs lock-free; right after the
    // probe saw no recovery artifacts, backend B truncates and is SIGKILLed after 1552.
    const a = new HistoryService(config);
    const tombstone = path.join(dir, "chat-archive.jsonl.truncate");
    const originalStat = fs.stat;
    let raced = false;
    const statSpy = spyOn(fs, "stat").mockImplementation((async (
      ...args: Parameters<typeof fs.stat>
    ) => {
      const run = () => originalStat(...args);
      if (raced || String(args[0]) !== tombstone) return run();
      raced = true;
      try {
        return await run();
      } finally {
        runCrashingForeignTruncation(config.rootDir, ws);
      }
    }) as typeof fs.stat);
    ok(await a.getHistoryFromLatestBoundary(ws)); // lazy rotation (2780) without recovery
    statSpy.mockRestore();
    expect(raced).toBe(true);

    const after = await fullIds(new HistoryService(config), ws); // next operation recovers
    // pre-0: removed by the crashed truncation or not (either is acceptable).
    expect(after.filter((id) => id !== "pre-0")).toEqual([
      "boundary-1",
      "post-0",
      "boundary-2",
      "post-1",
    ]);
  });
});

  test("B crashes before A starts reading: no rows are lost (passes today)", async () => {

    expect(before).toEqual(["pre-0", "boundary-1", "post-0", "boundary-2", "post-1"]);

    statSpy.mockRestore();

/** Backend B: a real second process that crashes after rename(archive -> tombstone), before chat.jsonl. */
function runCrashingForeignTruncation(rootDir: string, ws: string): void {
  const script = `
    import { Config } from "@/node/config";
    import { HistoryService } from "@/node/services/historyService";
    // writeFileAtomic calls the CommonJS fs.rename at call time (see its header), so this
    // hook sees the chat.jsonl publication (1565) and dies before it: the disk then holds the
    // marker (1550) and the tombstone (1552) with no archive, exactly a crash at that point.
    const nodeFs = require("fs");
    const rename = nodeFs.rename;
    nodeFs.rename = (from, to, callback) => {
      if (String(to).endsWith("chat.jsonl") && nodeFs.existsSync(String(to).slice(0, -"chat.jsonl".length) + "chat-archive.jsonl.truncate"))
        process.kill(process.pid, "SIGKILL");
      return rename(from, to, callback);
    };
    const result = await new HistoryService(new Config(${JSON.stringify(rootDir)})).truncateHistory(${JSON.stringify(ws)}, 0.1);
    console.log("NOT CRASHED", JSON.stringify(result));
  `;
  const child = spawnSync(process.execPath, ["--eval", script], {
    cwd: path.resolve(__dirname, "../../.."),
    encoding: "utf8",
    timeout: 60_000,
  });
  assert(child.signal === "SIGKILL", `backend B did not crash: ${child.stdout}${child.stderr}`);
}
